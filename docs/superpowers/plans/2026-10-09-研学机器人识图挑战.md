```python
#!/usr/bin/env python3
"""Generate data/robots.js from the source xlsx.

Usage: python tools/gen_robots.py <xlsx> <out_js>
Sheet1 columns: seq | brand | model | year | country | url
"""
import sys
import json
import openpyxl


def main():
    xlsx, out_js = sys.argv[1], sys.argv[2]
    wb = openpyxl.load_workbook(xlsx, data_only=True)
    ws = wb['Sheet1']
    robots = []
    for row in ws.iter_rows(min_row=2, values_only=True):
        seq = row[0]
        if seq is None:
            continue
        robots.append({
            'no': str(int(seq)),
            'brand': (row[1] or '').strip(),
            'model': (row[2] or '').strip(),
            'year': int(row[3]) if row[3] is not None else None,
            'country': (row[4] or '').strip(),
        })
    body = json.dumps(robots, ensure_ascii=False, indent=2)
    js = (
        '/* Generated by tools/gen_robots.py. Do not edit by hand. */\n'
        '(function (global) {\n'
        '  "use strict";\n'
        '  const ROBOTS = ' + body + ';\n'
        '  if (typeof module !== "undefined" && module.exports) module.exports = ROBOTS;\n'
        '  global.ROBOTS = ROBOTS;\n'
        '})(typeof window !== "undefined" ? window : this);\n'
    )
    with open(out_js, 'w', encoding='utf-8') as f:
        f.write(js)
    print('wrote', len(robots), 'robots to', out_js)


if __name__ == '__main__':
    main()
```

```python
#!/usr/bin/env python3
"""Compress robot photos into <photos_dir>/N.jpg and write photos.js.

Usage: python tools/compress_photos.py <src_dir> <photos_dir> <out_js>
A real photo filename has at least two dots, e.g.
'24. Unitree [X] 2025 CN.png' or '127..png'.
Single-dot names like '1.jpg' are group covers and are skipped.
"""
import os
import re
import sys
from PIL import Image

PAT = re.compile(r'^(\d+)\..*\.\w+$')
MAX = 1200
QUALITY = 82


def main():
    src, photos_dir, out_js = sys.argv[1], sys.argv[2], sys.argv[3]
    os.makedirs(photos_dir, exist_ok=True)
    by_no = {}
    for root, _dirs, files in os.walk(src):
        for name in files:
            m = PAT.match(name)
            if m:
                by_no[int(m.group(1))] = os.path.join(root, name)
    nums = sorted(by_no)
    for no in nums:
        img = Image.open(by_no[no]).convert('RGB')
        img.thumbnail((MAX, MAX))
        img.save(os.path.join(photos_dir, '%d.jpg' % no), 'JPEG',
                 quality=QUALITY, optimize=True)
    with open(out_js, 'w', encoding='utf-8') as f:
        f.write('/* Generated by tools/compress_photos.py. Do not edit by hand. */\n')
        f.write('(function (global) {\n  "use strict";\n  const PHOTO_LIST = [\n')
        for no in nums:
            f.write('    "%d.jpg",\n' % no)
        f.write('  ];\n')
        f.write('  if (typeof module !== "undefined" && module.exports) module.exports = PHOTO_LIST;\n')
        f.write('  global.PHOTO_LIST = PHOTO_LIST;\n')
        f.write('})(typeof window !== "undefined" ? window : this);\n')
    print('wrote', len(nums), 'photos; range', nums[0], '-', nums[-1])


if __name__ == '__main__':
    main()
```

```js
(function (global) {
  'use strict';

  function shuffle(arr) {
    const a = arr.slice();
    for (let i = a.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [a[i], a[j]] = [a[j], a[i]];
    }
    return a;
  }

  // 1 个正确 + distractCount 个随机干扰，打乱顺序
  function makeOptions(correct, all, distractCount = 3) {
    const others = all.filter(r => r.no !== correct.no);
    const wrong = shuffle(others).slice(0, distractCount);
    return shuffle([correct, ...wrong]);
  }

  // 照片清单 → 机型，按 no 匹配，匹配不到的跳过，保持照片与题目对齐
  function buildOrder(photoList, robots) {
    const byNo = new Map(robots.map(r => [r.no, r]));
    const order = [];
    const photoFiles = [];
    for (const file of photoList) {
      const no = file.replace(/\.[^.]+$/, '');
      const robot = byNo.get(no);
      if (robot) {
        order.push(robot);
        photoFiles.push(file);
      }
    }
    return { order, photoFiles };
  }

  // 随机抽 n 题，order 与 photoFiles 一一对应
  function pickRound(order, photoFiles, n) {
    const idx = shuffle(order.map((_, i) => i)).slice(0, n);
    return {
      order: idx.map(i => order[i]),
      photoFiles: photoFiles ? idx.map(i => photoFiles[i]) : null,
    };
  }

  const Quiz = { shuffle, makeOptions, buildOrder, pickRound };
  if (typeof module !== 'undefined' && module.exports) module.exports = Quiz;
  global.Quiz = Quiz;
})(typeof window !== 'undefined' ? window : this);
```

```html
    (function () {
      'use strict';
      const $ = id => document.getElementById(id);
      const ROUND = 10;
      const FLASH_MS = 650;

      const state = {
        order: [], photoFiles: [], index: 0, correctCount: 0,
        times: [], qStart: 0, roundStart: 0, correctIndex: -1, locked: false,
      };
      let timerId = null;

      function show(id) {
        document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
        $(id).classList.add('active');
      }

      const fmt = sec => sec.toFixed(1);

      function tickTimer() {
        $('q-timer').textContent = fmt((performance.now() - state.roundStart) / 1000) + ' 秒';
      }

      function startRound() {
        const { order, photoFiles } = Quiz.buildOrder(PHOTO_LIST, ROBOTS);
        const round = Quiz.pickRound(order, photoFiles, ROUND);
        state.order = round.order;
        state.photoFiles = round.photoFiles;
        state.index = 0;
        state.correctCount = 0;
        state.times = [];
        state.locked = false;
        state.roundStart = performance.now();
        show('screen-quiz');
        if (timerId) clearInterval(timerId);
        timerId = setInterval(tickTimer, 200);
        tickTimer();
        renderQuestion();
      }

      function renderQuestion() {
        const robot = state.order[state.index];
        $('q-progress').textContent = `第 ${state.index + 1}/${state.order.length} 题`;
        $('q-bar').style.width = (state.index / state.order.length * 100) + '%';

        const img = $('quiz-photo'), ph = $('photo-placeholder');
        img.style.display = 'none';
        ph.style.display = 'none';
        img.onload = () => { img.style.display = 'block'; ph.style.display = 'none'; };
        img.onerror = () => { img.style.display = 'none'; ph.style.display = 'flex'; };
        img.src = 'photos/' + state.photoFiles[state.index];

        const opts = Quiz.makeOptions(robot, ROBOTS, 3);
        state.correctIndex = opts.findIndex(o => o.no === robot.no);
        const box = $('q-options');
        box.innerHTML = '';
        const letters = ['A', 'B', 'C', 'D'];
        opts.forEach((o, i) => {
          const btn = document.createElement('button');
          btn.className = 'option';
          btn.innerHTML = '<span class="letter">' + letters[i] + '</span>' +
            '<span class="txt"><span class="m">' + o.model + '</span><span class="b">' + o.brand + '</span></span>';
          btn.addEventListener('click', () => answer(i));
          box.appendChild(btn);
        });
        state.qStart = performance.now();
      }

      function answer(i) {
        if (state.locked) return;
        state.locked = true;
        const buttons = document.querySelectorAll('.option');
        buttons.forEach(b => { b.disabled = true; });
        const isRight = i === state.correctIndex;
        buttons[i].classList.add(isRight ? 'correct' : 'wrong');
        if (!isRight) buttons[state.correctIndex].classList.add('correct');
        buttons.forEach((b, k) => { if (k !== i && k !== state.correctIndex) b.classList.add('dim'); });
        state.times.push((performance.now() - state.qStart) / 1000);
        if (isRight) state.correctCount += 1;
        setTimeout(next, FLASH_MS);
      }

      function next() {
        state.locked = false;
        state.index += 1;
        if (state.index >= state.order.length) showResult();
        else renderQuestion();
      }

      function showResult() {
        if (timerId) { clearInterval(timerId); timerId = null; }
        const n = state.order.length;
        const total = state.times.reduce((a, b) => a + b, 0);
        $('r-time').textContent = fmt(total);
        $('r-correct').textContent = state.correctCount + '/' + n;
        $('r-acc').textContent = Math.round(state.correctCount / n * 100) + '%';
        $('r-avg').textContent = fmt(total / n);
        $('r-praise').textContent = praise(state.correctCount, n);
        show('screen-result');
      }

      function praise(correct, n) {
        const r = correct / n;
        if (r === 1) return '满分！你就是机器人识别大师 🏆';
        if (r >= 0.8) return '太厉害了，眼神超准 👏';
        if (r >= 0.5) return '不错哦，再快一点就更棒啦！';
        return '别灰心，再看一遍照片墙，你可以的 💪';
      }

      $('btn-start').addEventListener('click', startRound);
      $('btn-restart').addEventListener('click', startRound);
    })();
  </script>
```
