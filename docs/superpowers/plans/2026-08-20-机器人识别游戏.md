# 机器人识别游戏 实现计划

> **针对 agent 工作者：** 必需的子 skill：使用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans 来逐个任务执行本计划。步骤使用复选框（`- [ ]`）语法进行追踪。

**目标：** 做一个"看图识机型"的四选一网页小游戏：显示机器人照片，玩家从 4 个选项中选出正确的品牌+型号，全部答完显示总分。

**架构：** 纯前端单页应用，无需构建工具，双击 index.html 即可玩。数据（182 台机型）从 Excel 导出为 `data/robots.js`；照片顺序清单在 `data/photos.js`；纯逻辑（洗牌/生成选项）放在 `js/quiz.js`（浏览器和 Node 双端可用，便于单测）；界面与游戏流程在 `index.html`。

**技术栈：** HTML + CSS + 原生 JavaScript（无框架、无依赖）；测试用 Node.js 内置的 `node:test`（node v24 已装，无需 npm install）。

**设计文档：** `docs/superpowers/specs/2026-08-20-机器人识别游戏-design.md`

## 文件结构

```
robot-game/
├── index.html            ← 界面 + 游戏流程（HTML/CSS/JS）
├── js/
│   └── quiz.js           ← 纯逻辑：shuffle、makeOptions（浏览器 + Node 通用）

│   └── photos.js         ← 照片顺序清单（window.PHOTO_LIST / module.exports）
├── photos/               ← 机器人照片目录（含 .gitkeep 占位）
└── test/
    ├── robots.test.js    ← 数据完整性测试
    └── quiz.test.js      ← 纯逻辑测试
```

## 数据约定

- Excel 导出到 `data/robots.js`，每台字段：`{ no, brand, model, category, subcategory, feature }`。`feature`（一句话特点）可能为空字符串，界面需兜底。
- `data/photos.js` 的 `PHOTO_LIST` 是**文件名数组**，按用户照片顺序排列，如 `['NO.001.jpg', 'NO.002.png']`。游戏按此顺序出题，只包含能在数据库里匹配到编号的照片。空数组 = 还没有照片。
- 照片文件放 `photos/` 目录，文件名 = 编号 + 扩展名（如 `NO.001.jpg`）。

创建 `js/quiz.js`：

```js
(function (global) {
  'use strict';

  function shuffle(arr) {
    const a = arr.slice();
    for (let i = a.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [a[i], a[j]] = [a[j], a[i]];
    }
    return a;
  }

  // 生成 4 个选项：1 个正确 + distractCount 个全随机干扰，打乱顺序
  function makeOptions(correct, all, distractCount = 3) {
    const others = all.filter(r => r.no !== correct.no);
    const wrong = shuffle(others).slice(0, distractCount);
    return shuffle([correct, ...wrong]);
  }

  const Quiz = { shuffle, makeOptions };
  if (typeof module !== 'undefined' && module.exports) module.exports = Quiz;
  global.Quiz = Quiz;
})(typeof window !== 'undefined' ? window : this);
```

把 `<script id="game-script"></script>` 替换为：

```html
  <script>
    (function () {
      'use strict';
      const $ = id => document.getElementById(id);
      const TRIAL_COUNT = 10;

      const state = { order: [], index: 0, score: 0, correctCount: 0 };

      function show(id) {
        document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
        $(id).classList.add('active');
      }

      // 根据照片清单构建题目顺序；无照片返回 null
      function loadOrder() {
        if (!PHOTO_LIST.length) return null;
        const byNo = new Map(ROBOTS.map(r => [r.no, r]));
        const order = [];
        for (const file of PHOTO_LIST) {
          const no = file.replace(/\.[^.]+$/, '');
          const robot = byNo.get(no);
          if (robot) order.push(robot);
        }
        return order;
      }

      // 试玩：随机抽 10 台
      function trialOrder() {
        return Quiz.shuffle(ROBOTS).slice(0, TRIAL_COUNT);
      }

      function renderStatus() {
        $('q-progress').textContent = `第 ${state.index + 1}/${state.order.length} 题`;
        $('q-score').textContent = state.score;
        $('q-bar').style.width = (state.index / state.order.length * 100) + '%';
      }

      function renderPhoto(robot) {
        const img = $('quiz-photo');
        const ph = $('photo-placeholder');
        $('ph-no').textContent = robot.no;
        $('ph-model').textContent = robot.model;
        if (state.photoFile) {
          img.onload = () => { img.style.display = 'block'; ph.style.display = 'none'; };
          img.onerror = () => { img.style.display = 'none'; ph.style.display = 'flex'; };
          img.src = 'photos/' + state.photoFile;
        } else {
          img.style.display = 'none';
          ph.style.display = 'flex';
        }
      }

      function renderQuestion() {
        const robot = state.order[state.index];
        state.photoFile = state.photoFiles ? state.photoFiles[state.index] : '';
        state.correctIndex = -1;
        $('feedback').className = 'feedback';
        renderStatus();
        renderPhoto(robot);

        const opts = Quiz.makeOptions(robot, ROBOTS);
        state.correctIndex = opts.findIndex(o => o.no === robot.no);

        const box = $('q-options');
        box.innerHTML = '';
        const letters = ['A', 'B', 'C', 'D'];
        opts.forEach((o, i) => {
          const btn = document.createElement('button');
          btn.className = 'option';
          btn.innerHTML =
            '<span class="letter">' + letters[i] + '</span>' +
            '<span class="txt"><span class="m">' + o.model + '</span><span class="b">' + o.brand + '</span></span>';
          btn.addEventListener('click', () => answer(i));
          box.appendChild(btn);
        });
      }

      function answer(i) {
        const buttons = document.querySelectorAll('.option');
        buttons.forEach(b => { b.disabled = true; });
        const isRight = i === state.correctIndex;
        buttons[i].classList.add(isRight ? 'correct' : 'wrong');
        if (!isRight) buttons[state.correctIndex].classList.add('correct');
        buttons.forEach((b, k) => { if (k !== i && k !== state.correctIndex) b.classList.add('dim'); });

        const robot = state.order[state.index];
        if (isRight) {
          state.score += 10;
          state.correctCount += 1;
        }
        renderStatus();
        showFeedback(isRight, robot);
      }

      function showFeedback(isRight, robot) {
        const fb = $('feedback');
        fb.className = 'feedback ' + (isRight ? 'right' : 'wrong');
        $('fb-title').textContent = isRight ? '✓ 答对了！' : '✗ 答错了';
        $('fb-ans').textContent = '正确答案：' + robot.brand + ' ' + robot.model;
        $('fb-feature').textContent = robot.feature ? robot.feature : '（暂无一句话特点）';
        $('btn-next').textContent = state.index + 1 >= state.order.length ? '查看结果' : '下一题';
      }

      function next() {
        state.index += 1;
        if (state.index >= state.order.length) {
          showResult();
        } else {
          renderQuestion();
        }
      }

      function showResult() {
        $('r-score').textContent = state.score;
        $('r-sub').textContent = `共 ${state.order.length} 题，答对 ${state.correctCount} 题`;
        show('screen-result');
      }

      function startQuiz(order, photoFiles) {
        state.order = order;
        state.photoFiles = photoFiles || null;
        state.index = 0;
        state.score = 0;
        state.correctCount = 0;
        show('screen-quiz');
        renderQuestion();
      }

      $('btn-start').addEventListener('click', () => {
        const order = loadOrder();
        if (!order || !order.length) {
          show('screen-setup');
        } else {
          startQuiz(order, PHOTO_LIST);
        }
      });

      $('btn-try').addEventListener('click', () => startQuiz(trialOrder(), null));
      $('btn-back').addEventListener('click', () => show('screen-start'));
      $('btn-next').addEventListener('click', next);
      $('btn-restart').addEventListener('click', () => show('screen-start'));
    })();
  </script>
```
